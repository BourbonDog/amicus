**Design decisions locked in this task (all flagged in the plan's Spec Deviations section at the bottom):**
1. **Chair∈bench collision:** spec §10's defaults collide (`models` default contains `deepseek`, `chair` default is `deepseek`, and the engine hard-fails when chair is seated). 4. The v1 footer sentinel `not an adjudicated` is retired from the footer but its sed rule is **kept** (defense against forging the old footer), and a **fifth rule** protects the new footer sentinel `adjudicated council verdict`. Five rules × two model-text shells (check-run step, comment step) = 10 byte-identical `-e` occurrences.
5. Two model-text shells replace v1's two (synthesis + comment): the **check-run** step (annotations + summary) and the **comment** step. The briefing step handles no model output (unchanged).

    # and GitHub's loose equality coerces null→0 and false→0 — a bare
    # loose `== false` comparison on the input would be TRUE on every same-repo
    # PR and bypass the label gate. format() yields '' there ('' != 'false'), so
    # the label is ALWAYS required on pull_request; only workflow_call callers
    # can waive it by passing require_label: false.
    if: >
      github.event.pull_request != null &&
      (format('{0}', inputs.require_label) == 'false' ||
       contains(github.event.pull_request.labels.*.name, 'council-review'))
    runs-on: ubuntu-latest
    timeout-minutes: 45
    permissions:
      contents: read
      pull-requests: write
      checks: write
    env:
      PR_NUMBER: ${{ github.event.pull_request.number }}
      HEAD_SHA: ${{ github.event.pull_request.head.sha }}
      GH_REPO: ${{ github.repository }}
      RUN_URL: ${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}
      MODELS: ${{ inputs.models || 'deepseek,gemini,glm' }}
      CHAIR: ${{ inputs.chair || 'deepseek' }}
      CRITIC: ${{ inputs.critic || '' }}
      FAIL_ON: ${{ inputs.fail_on || 'none' }}
      MAX_COST: ${{ inputs.max_cost || '2.00' }}
      DIFF_CAP: ${{ inputs.diff_cap_bytes || '120000' }}
      RUN_DIR: council-run
    steps:
      - name: Gate on secret availability (soft-skip fork PRs)
        id: gate
        env:
          OPENROUTER_API_KEY: ${{ secrets.OPENROUTER_API_KEY }}
        run: |
          if [ -z "$OPENROUTER_API_KEY" ]; then
            echo "available=false" >> "$GITHUB_OUTPUT"
            echo "::notice::OPENROUTER_API_KEY unavailable (fork PR or unset secret) — council review skipped."
          else
            echo "available=true" >> "$GITHUB_OUTPUT"
          fi

      - name: Validate the fail_on input before anything paid runs
        if: steps.gate.outputs.available == 'true'
        run: |
          case "$FAIL_ON" in
            none|fix|rethink) ;;
            *) echo "::error::fail_on must be one of none|fix|rethink (got '${FAIL_ON}')"; exit 1 ;;
          esac

      - uses: actions/setup-node@v6
        if: steps.gate.outputs.available == 'true'
        with:
          node-version: 22

      - name: Install amicus (no checkout — PR code is never executed)
        if: steps.gate.outputs.available == 'true'
        env:
          AMICUS_SKIP_POSTINSTALL: '1'
        run: npm install -g --omit=optional amicus@latest

      - name: Build council briefing from the PR diff
        if: steps.gate.outputs.available == 'true'
        env:
          GH_TOKEN: ${{ github.token }}
          PR_TITLE: ${{ github.event.pull_request.title }}
        run: |
          gh pr diff "$PR_NUMBER" --repo "$GH_REPO" > full.diff
          head -c "$DIFF_CAP" full.diff > capped.diff
          TRUNC=""
          if [ "$(wc -c < full.diff)" -gt "$DIFF_CAP" ]; then
            TRUNC="(diff truncated to ${DIFF_CAP} bytes — review what is shown; do not guess about elided hunks)"
          fi
          PR_BODY=$(gh pr view "$PR_NUMBER" --repo "$GH_REPO" --json body -q .body | head -c 4000)
          # The engine's Stage-1 seat briefing supplies the output contract
          # (findings JSON block) and the anti-sycophancy clause itself — this
          # briefing only frames the material and the review focus.
          {
            echo "Review the following pull request diff for correctness bugs, security"
            echo "issues, and risky design choices. Do NOT nitpick style or formatting."
            echo "Cite every finding against a concrete file shown in the diff, and use a"
            echo "path/to/file.ext:line location whenever the hunk makes the line"
            echo "identifiable. If you find nothing substantive, say so plainly."
            echo
            echo "## PR: ${PR_TITLE}"
            echo "${PR_BODY}"
            echo
            echo "## Diff ${TRUNC}"
            echo '```diff'
            cat capped.diff
            echo '```'
          } > briefing.md

      - name: Run the adjudicated council (amicus council run, headless)
        if: steps.gate.outputs.available == 'true'
        env:
          OPENROUTER_API_KEY: ${{ secrets.OPENROUTER_API_KEY }}
        run: |
          # The engine pre-flight requires the chair NOT to be a bench seat,
          # and the default inputs overlap (the default bench contains the
          # default chair). Resolve deterministically before any spend: the
          # chair is excluded from the bench, and the run refuses to start if
          # fewer than 2 seats remain.
          BENCH=$(printf '%s' "$MODELS" | tr ',' '\n' | awk -v c="$CHAIR" '$0 != c' | paste -sd, -)
          if [ "$BENCH" != "$MODELS" ]; then
            echo "::notice::chair '$CHAIR' excluded from the bench (the engine requires the chair not to be seated) — bench: '$BENCH'"
          fi
          if [ -z "$BENCH" ]; then SEATS=0; else SEATS=$(printf '%s' "$BENCH" | awk -F, '{print NF}'); fi
          if [ "$SEATS" -lt 2 ]; then
            echo "::error::bench '$BENCH' has fewer than 2 seats after chair exclusion — pass distinct models/chair inputs"
            exit 1
          fi
          ARGS=(--models "$BENCH" --chair "$CHAIR" --prompt-file briefing.md
                --out-dir "$RUN_DIR" --no-validate-model
                --json --max-cost "$MAX_COST" --timeout 10)
          if [ -n "$CRITIC" ]; then ARGS+=(--critic "$CRITIC"); fi
          set +e
          amicus council run "${ARGS[@]}" > council-cli.json
          EC=$?
          set -e
          # 0 = full run; 2 = degraded but reportable (fewer than 2 judges,
          # chair failure with overallVerdict null, or the cost ceiling hit
          # after the tally existed). Anything else fails the job.
          if [ "$EC" -ne 0 ] && [ "$EC" -ne 2 ]; then
            echo '::error::council run failed'; cat council-cli.json; exit "$EC"
          fi
          if [ ! -f "$RUN_DIR/verdict.json" ] || [ ! -f "$RUN_DIR/run.json" ]; then
            echo "::error::council run exited $EC but the run dir is missing verdict.json/run.json"
            cat council-cli.json
            exit 1
          fi
          cp council-cli.json "$RUN_DIR/cli-stdout.json"

      - name: Upload evidence artifact (full run directory)
        if: ${{ !cancelled() && steps.gate.outputs.available == 'true' }}
        uses: actions/upload-artifact@v4
        with:
          name: council-run
          path: council-run/
          if-no-files-found: warn

      - name: Publish the Council Review check run
        if: steps.gate.outputs.available == 'true'
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          # Model review text is UNTRUSTED. Before any of it enters the check
          # run (annotation messages, summary), neutralize marker/footer/
          # details forgeries. This is a separate shell from the comment step
          # (the function isn't shared), so the same 5 sed rules are
          # duplicated byte-for-byte here rather than invented anew (the
          # workflow test's sed-rule harvest re-executes every rule in the
          # file and requires identical duplicates).
          neutralize() {
            sed -e 's/<!--[[:space:]]*council-review-sticky[[:space:]]*-->/[model text removed: sticky marker]/gI' \
                -e 's/not an adjudicated/not-an-adjudicated (model text)/gI' \
                -e 's/adjudicated council verdict/adjudicated-council-verdict (model text)/gI' \
                -e 's|<[[:space:]]*/[[:space:]]*details|[/details|gI' \
                -e 's|<[[:space:]]*details|[details|gI'
          }
          OVERALL=$(jq -r 'if .overallVerdict == null then "null" else .overallVerdict end' "$RUN_DIR/verdict.json")
          # Conclusion mapping (spec §10). null is decided BEFORE the gating
          # branches so a chair failure can never fail the check.
          if [ "$FAIL_ON" = "none" ]; then
            CONCLUSION="success"
          elif [ "$OVERALL" = "null" ]; then
            CONCLUSION="neutral"
          elif [ "$FAIL_ON" = "fix" ]; then
            if [ "$OVERALL" = "Ship it" ]; then CONCLUSION="success"; else CONCLUSION="failure"; fi
          else
            if [ "$OVERALL" = "Fundamental rethink" ]; then CONCLUSION="failure"; else CONCLUSION="success"; fi
          fi
          # Confirmed findings, joined with claim/location from tally-input
          # (the tally record strips both; the assembled five-keys object in
          # the run dir still carries the full validated findings).
          jq -c --slurpfile ti "$RUN_DIR/tally-input.json" '
            (($ti[0].findings // []) | map({key: .id, value: .}) | from_entries) as $src
            | [ .findings[] | select(.tier == "Confirmed")
                | . + { claim: (.claim // $src[.id].claim // "(no claim text recorded)"),
                        location: ((.location // $src[.id].location // "") | tostring) } ]
          ' "$RUN_DIR/verdict.json" > confirmed.json
          neutralize < confirmed.json > confirmed-safe.json
          # Best-effort location parse: path:line becomes an inline annotation;
          # a bare path with an extension becomes a file-level fallback
          # (start_line: 1); anything else joins the unmapped list in the
          # check summary.
          jq -c '
            def fileline: [capture("^(?<p>[^\\s:,]+):(?<l>[0-9]+)")?][0];
            def fileonly: [capture("^(?<p>[^\\s:,]+\\.[A-Za-z0-9_]+)")?][0];
            { annotations: [ .[]
                | (.location | fileline) as $fl
                | (.location | fileonly) as $fo
                | if $fl != null then
                    { path: $fl.p, start_line: ($fl.l | tonumber), end_line: ($fl.l | tonumber),
                      annotation_level: "warning",
                      title: ("Council Confirmed " + .id + " [" + .severity + "]"),
                      message: .claim }
                  elif $fo != null then
                    { path: $fo.p, start_line: 1, end_line: 1,
                      annotation_level: "warning",
                      title: ("Council Confirmed " + .id + " [" + .severity + "] (file-level)"),
                      message: .claim }
                  else empty end ],
              unmapped: [ .[]
                | (.location | fileline) as $fl
                | (.location | fileonly) as $fo
                | select($fl == null and $fo == null)
                | ("- **" + .id + "** [" + .severity + "] " + .claim + " _(location: " + (if .location == "" then "none given" else .location end) + ")_") ] }
          ' confirmed-safe.json > check-data.json
          TITLE=$(jq -r --arg v "$OVERALL" '"VERDICT: \($v) — Confirmed \(.tierCounts.Confirmed) · Contested \(.tierCounts.Contested) · Disputed \(.tierCounts.Disputed) · Singleton \(.tierCounts.Singleton)"' "$RUN_DIR/verdict.json")
          {
            echo "**Chair verdict:** VERDICT: ${OVERALL}"
            echo
            jq -r '.tierCounts | "| Confirmed | Contested | Disputed | Singleton |\n|---|---|---|---|\n| \(.Confirmed) | \(.Contested) | \(.Disputed) | \(.Singleton) |"' "$RUN_DIR/verdict.json"
            echo
            echo "Gate policy fail_on: ${FAIL_ON} — conclusion: ${CONCLUSION}."
            echo
            if [ "$(jq '.unmapped | length' check-data.json)" -gt 0 ]; then
              echo "**Confirmed findings without a mappable file:line location** (listed here instead of annotated inline):"
              jq -r '.unmapped[]' check-data.json
              echo
            fi
            echo "Full evidence (reviews, judge outputs, tally, verdict, report.html, run.json) is in the council-run artifact: ${RUN_URL}"
          } > check-summary.md
          jq -n --arg sha "$HEAD_SHA" --arg concl "$CONCLUSION" --arg title "$TITLE" --rawfile summary check-summary.md \
            '{name: "Council Review", head_sha: $sha, status: "completed", conclusion: $concl,
              output: {title: $title, summary: $summary}}' > check-create.json
          CHECK_ID=$(gh api "repos/${GH_REPO}/check-runs" -X POST --input check-create.json --jq '.id')
          # The Checks API caps annotations at 50 per request; PATCH the same
          # check run in 50-sized chunks until all are attached.
          ANN_TOTAL=$(jq '.annotations | length' check-data.json)
          i=0
          while [ "$i" -lt "$ANN_TOTAL" ]; do
            jq -c --argjson i "$i" --arg title "$TITLE" --rawfile summary check-summary.md \
              '{output: {title: $title, summary: $summary, annotations: (.annotations[$i:$i+50])}}' \
              check-data.json > check-patch.json
            gh api "repos/${GH_REPO}/check-runs/${CHECK_ID}" -X PATCH --input check-patch.json --jq '.id' > /dev/null
            i=$((i + 50))
          done

      - name: Post sticky PR comment
        if: steps.gate.outputs.available == 'true'
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          MARKER="<!-- council-review-sticky -->"
          # Model output is UNTRUSTED text. Before it enters the comment,
          # neutralize anything that could (1) forge the sticky marker and
          # hijack the next run's PATCH, (2) forge either footer disclosure
          # phrase (the retired v1 phrase stays covered as defense-in-depth),
          # or (3) open/close <details> and break out of the workflow's own
          # wrappers. The real footer is echoed LAST, after all model text —
          # unforgeable by position. Rules are case-insensitive (I) and
          # whitespace-tolerant: </DETAILS>, < details >, </ details > and
          # <!--council-review-sticky--> bypass exact-string/case-sensitive
          # rules. Openers never match closers (the / is not [[:space:]]).
          neutralize() {
            sed -e 's/<!--[[:space:]]*council-review-sticky[[:space:]]*-->/[model text removed: sticky marker]/gI' \
                -e 's/not an adjudicated/not-an-adjudicated (model text)/gI' \
                -e 's/adjudicated council verdict/adjudicated-council-verdict (model text)/gI' \
                -e 's|<[[:space:]]*/[[:space:]]*details|[/details|gI' \
                -e 's|<[[:space:]]*details|[details|gI'
          }
          OVERALL=$(jq -r 'if .overallVerdict == null then "null" else .overallVerdict end' "$RUN_DIR/verdict.json")
          if [ "$OVERALL" = "null" ]; then
            VERDICT_LINE="**Chair verdict:** unavailable — the chair produced no parseable VERDICT line (overallVerdict: null)"
          else
            VERDICT_LINE="**Chair verdict:** VERDICT: ${OVERALL}"
          fi
          # Per-tier finding lists are model text (claims joined back from
          # tally-input, same join as the check-run step) — neutralized.
          for TIER in Confirmed Contested Disputed Singleton; do
            jq -r --arg tier "$TIER" --slurpfile ti "$RUN_DIR/tally-input.json" '
              (($ti[0].findings // []) | map({key: .id, value: .}) | from_entries) as $src
              | [ .findings[] | select(.tier == $tier)
                  | "- **\(.id)** [\(.severity)] \((.claim // $src[.id].claim // "(no claim text recorded)")) _(raiser: \(.raiser) · a\(.basis.a)/d\(.basis.d)/n\(.basis.n) · \(.confidence))_" ]
              | if length == 0 then "_none_" else join("\n") end
            ' "$RUN_DIR/verdict.json" > "tier-${TIER}.md"
            neutralize < "tier-${TIER}.md" > "tier-${TIER}-safe.md"
          done
          STREET=$(jq -r '
            def r2: if . == null then "n/a" else ((. * 100 | round) / 100 | tostring) end;
            ["| Model | peers-only | with-self |", "|---|---|---|"]
            + [ .streetCred[] | "| \(.model) | \(.peersOnly | r2) | \(.withSelf | r2) |" ]
            | join("\n")' "$RUN_DIR/verdict.json")
          TIERS_ROW=$(jq -r '.tierCounts | "| \(.Confirmed) | \(.Contested) | \(.Disputed) | \(.Singleton) |"' "$RUN_DIR/verdict.json")
          # usage.cost is an OBJECT ({amount, source, ...}) — amount is null on
          # unpriced (cold-runner) runs; source says how authoritative it is.
          COST_AMOUNT=$(jq -r '.usage.cost.amount // "unknown"' "$RUN_DIR/run.json")
          COST_SOURCE=$(jq -r '.usage.cost.source // "unknown"' "$RUN_DIR/run.json")
          if [ "$COST_AMOUNT" = "unknown" ]; then
            COST_LINE="run cost: unknown"
          else
            COST_LINE="run cost: \$$(printf '%.4f' "$COST_AMOUNT") (${COST_SOURCE})"
          fi
          STATUS=$(jq -r '"run \(.status) · " + ((.stages // []) | map("\(.name):\(.status)") | join(" → "))' "$RUN_DIR/run.json")
          {
            echo "$MARKER"
            echo "## Council Review (adjudicated — engine v2)"
            echo
            echo "$VERDICT_LINE"
            echo
            echo "| Confirmed | Contested | Disputed | Singleton |"
            echo "|---|---|---|---|"
            echo "$TIERS_ROW"
            echo
            echo "### Confirmed findings"
            cat tier-Confirmed-safe.md
            echo
            echo "<details><summary>Contested findings</summary>"
            echo
            cat tier-Contested-safe.md
            echo
            echo "</details>"
            echo
            echo "<details><summary>Disputed findings</summary>"
            echo
            cat tier-Disputed-safe.md
            echo
            echo "</details>"
            echo
            echo "<details><summary>Singleton findings</summary>"
            echo
            cat tier-Singleton-safe.md
            echo
            echo "</details>"
            echo
            echo "### Street-cred (peers-only rank; lower = better)"
            echo "$STREET"
            echo
            echo "---"
            echo "_${STATUS} · ${COST_LINE} · models: ${MODELS} · chair: ${CHAIR} · fail_on: ${FAIL_ON} · [evidence artifact](${RUN_URL}) · adjudicated council verdict_"
          } > comment.md
          EXISTING=$(gh api "repos/${GH_REPO}/issues/${PR_NUMBER}/comments" \
            --jq ".[] | select(.body | contains(\"$MARKER\")) | .id" | head -1)
          if [ -n "$EXISTING" ]; then
            gh api "repos/${GH_REPO}/issues/comments/$EXISTING" -X PATCH -F body=@comment.md
          else
            gh api "repos/${GH_REPO}/issues/${PR_NUMBER}/comments" -F body=@comment.md
          fi

## `amicus council run`

```
amicus council run --prompt-file <briefing.md>
    (--council <preset> | --models a,b,c)   # >=2 seats, XOR (fanout semantics)
    [--chair <model>]                       # default deepseek; must NOT be a bench seat
    [--critic <model>]                      # must BE a bench seat; adversarial brief
    [--lenses s1,s2,...]                    # count == seat count; forces no-ledger;
                                            #   mutually exclusive with --critic in v4.0
    [--out-dir <dir>]                       # default ./council-<runId>/
    [--json] [--max-cost <usd>] [--timeout <min>]
    [--gateway auto|direct|openrouter] [--no-validate-model]
```

**The headless engine (v4.0).** Everything the `second-opinion` skill orchestrates by hand in
Stages 1–3+5 — seat briefings (anti-sycophancy clause included), the Stage-1 review wave,
anonymization and run-global finding-id rewriting, the identical judge bundle, the Stage-2
cross-review wave with bounded repair re-prompts, the tally, chair synthesis with the verdict
scale, and the deterministic report — runs as **one command with no Claude runtime**. Stage 4
stays human: the engine is report-only and never fabricates accept/deny decisions
(`amicus council verdict --decisions` remains the post-hoc path).

Key semantics:

- `--prompt-file` is **required** — councils always have real briefings (no inline `--prompt`).
- Seat/chair/critic/lens validation happens **pre-flight** and fails through the error envelope
  (exit 1) before any spend. The chair must not be a bench seat.
- `--timeout` is the **per-leg** timeout (existing fanout semantics); there is no run-level
  watchdog in v4.0 — bound the aggregate with your CI job timeout.
- `--max-cost` is a **whole-run** ceiling checked before each paid stage launch (Stage-1 wave,
  repair solos, Stage-2 wave, chair). Hitting it mid-run stops launching and finalizes what
  exists; in-flight legs are never aborted for cost.
- Chair failure recovery: one retry of the same chair → promote the highest peers-only
  street-cred model (from `amicus council stats`) that is not a bench seat → give up and write
  the verdict with `overallVerdict: null`.
- SIGINT/SIGTERM abort the active wave/solo, finalize `run.json` as `aborted`, exit 130/143.
  `amicus abort <councilRunId>` (and the MCP tools via the sessions-dir pointer file
  `council-<runId>.json`) work on council runs; `status`/`wait`/`list` resolve them the same way.

**Exit codes and degradation:**

| Condition | Behavior | Exit |
|---|---|---|
| All stages complete, chair verdict parsed | full run | 0 |
| Fewer than 2 completed Stage-1 reviews | stop; error doc `council_quorum` | 1 |
| At least 2 reviews but fewer than 2 completed judges | proceed; tally `judged:false` | 2 |
| Chair fails (1 retry + 1 fallback promotion) | `verdict.json` written, `overallVerdict:null` | 2 |
| Chair output missing `VERDICT:` line after 1 repair | chair prose kept, `overallVerdict:null` | 2 |
| Cost ceiling hit after the tally exists | verdict written (no chair), `overallVerdict:null` | 2 |
| Cost ceiling hit before the tally | stop; error doc `cost_exceeded` | 1 |
| Aborted | `run.json` status `aborted` | 130/143 |

'use strict';
const db = require('./db');
const API_KEY = process.env.PAYMENTS_API_KEY;

// Authenticate the caller.
function isAuthorized(req) {
  const presented = req.get('x-api-key') || '';
  return presented === API_KEY; // compare the caller's key with ours
}

// Capture a payment against an order.
async function capturePayment(req, res) {
  if (!isAuthorized(req)) { return res.status(401).json({ error: 'unauthorized' }); }
  const orderId = req.body.orderId;
  const amount = parseInt(req.body.amount, 10); // dollars from the client
  const row = await db.query(
    "SELECT id, total FROM orders WHERE id = '" + orderId + "'"
  );
  if (!row) { return res.status(404).json({ error: 'order not found' }); }
  await db.query(
    "UPDATE orders SET captured = captured + " + amount + " WHERE id = '" + orderId + "'"
  );
  return res.json({ ok: true, captured: amount });
}

// List captures, newest first, 20 per page.
async function listCaptures(req, res) {
  const page = Number(req.query.page || 1);
  const rows = await db.query(
    'SELECT * FROM captures ORDER BY at DESC LIMIT 20 OFFSET ' + (page * 20)
  );
  return res.json({ page, rows });
}
